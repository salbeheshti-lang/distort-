     1→// DISTORT — Client-side recovery engine
     2→//
     3→// THE ACTUAL RECOVERY LOOP. Runs entirely in the browser/app.
     4→//
     5→// Per the user's design (§3):
     6→//   Start → Diagnose network → Gather options (local + remote) →
     7→//   Validate + prioritize → Connect with real engine → Verify →
     8→//   (success: monitor health) | (fail: try next or report no path)
     9→//
    10→// CRITICAL HONESTY RULES:
    11→//   1. "connected" is ONLY set after a real HTTP request succeeds through
    12→//      the tunnel AND the exit IP differs from the real IP.
    13→//   2. If no path works, state goes to "exhausted" with a clear reason —
    14→//      never a fake "connected".
    15→//   3. The server's plan is a SUGGESTION — the client does its own probing.
    16→//   4. No credentials are ever logged (use scrubSensitive on any string
    17→//      before displaying or persisting).
    18→//
    19→// BROWSER LIMITATIONS (documented honestly):
    20→//   - Cannot spawn xray-core (needs native process)
    21→//   - Cannot open raw TCP sockets
    22→//   - Cannot set system proxy (only chrome.proxy for the extension)
    23→//   - CAN: diagnose network, validate config structure, fetch plan from API,
    24→//     and — IF the desktop companion is running — delegate the actual
    25→//     tunnel to it via the local SOCKS5 proxy at 127.0.0.1:10808.
    26→//
    27→// DESKTOP COMPANION BRIDGE:
    28→//   When the Distort Desktop app is running, it exposes a local SOCKS5
    29→//   proxy on 127.0.0.1:10808. The browser can route fetch requests through
    30→//   it (via a PAC script or chrome.proxy.settings). If we detect the
    31→//   companion is up, we can do real tunnel verification from the browser
    32→//   by fetching through the SOCKS5 proxy and comparing the exit IP.
    33→//
    34→// WITHOUT DESKTOP COMPANION:
    35→//   The browser cannot actually establish a VPN tunnel. In this case, the
    36→//   engine diagnoses the network, lists available configs, but reports
    37→//   honestly: "Install the desktop or Android app to actually connect.
    38→//   The web dashboard can only diagnose and validate configs."

    53→// ---------------------------------------------------------------------------
    54→// Tunnel engine adapter (the abstraction over desktop/android/browser)
    55→// ---------------------------------------------------------------------------
    56→
    57→export type TunnelEngineResult = {
    58→  // Whether the engine successfully started the tunnel
    59→  started: boolean;
    60→  // Whether a real HTTP request through the tunnel succeeded
    61→  trafficVerified: boolean;
    62→  // Exit IP discovered through the tunnel (null if not verified)
    63→  exitIp: string | null;
    64→  // Real IP (for leak detection — if exitIp == realIp, traffic isn't going through tunnel)
    65→  realIp: string | null;
    66→  // Error reason if failed (NO credentials)
    67→  error: string | null;
    68→  // Latency through the tunnel in ms
    69→  latencyMs: number | null;
    70→};
    71→
    72→// The tunnel engine adapter. Different implementations for different platforms:
    73→//   - BrowserWithoutCompanion: cannot connect at all — honest "install desktop app"
    74→//   - BrowserWithCompanion: delegates to local SOCKS5 (127.0.0.1:10808)
    75→//   - Desktop (Electron): spawns xray-core directly
    76→//   - Android: starts VpnService with libxray.so
    77→export interface TunnelEngine {
    78→  // Platform name for logging
    79→  readonly platform: string;
    80→  // Whether this engine can actually establish tunnels
    81→  readonly canConnect: boolean;
    82→  // Start a tunnel with the given config URI + verify with real traffic
    83→  connect(configUri: string, timeoutMs: number): Promise<TunnelEngineResult>;
    84→  // Stop the current tunnel
    85→  disconnect(): Promise<void>;
    86→  // Check if the engine is currently running
    87→  isRunning(): boolean;
    88→}

   118→// ---------------------------------------------------------------------------
   119→// Browser + desktop companion engine (delegates to local SOCKS5)
   120→// ---------------------------------------------------------------------------
   121→
   122→export class BrowserWithCompanionEngine implements TunnelEngine {
   123→  readonly platform = "browser-with-companion";
   124→  readonly canConnect = true;
   125→  private running = false;
   126→
   127→  async connect(configUri: string, timeoutMs: number): Promise<TunnelEngineResult> {
   128→    // The companion app (Distort Desktop) is running and exposes:
   129→    //   - SOCKS5 at 127.0.0.1:10808
   130→    //   - A control API at 127.0.0.1:10809 (HTTP)
   131→    //
   132→    // Steps:
   133→    //   1. POST the configUri to the companion's control API
   134→    //      (it spawns xray with that config)
   135→    //   2. Wait for the companion to confirm xray started
   136→    //   3. Fetch exit IP through the SOCKS5 proxy
   137→    //   4. Fetch real IP directly (bypassing proxy)
   138→    //   5. Compare — if different, tunnel is verified
   139→    //
   140→    // BROWSER LIMITATION: fetch() cannot use a SOCKS5 proxy directly.
   141→    // We'd need chrome.proxy.settings (extension only) or a custom
   142→    // SOCKS5-over-WebSocket bridge. For the extension, this works.
   143→    // For the plain web app, we can only do the control API call.
   144→    //
   145→    // So in practice, this engine works ONLY in the Chrome extension context
   146→    // (which has chrome.proxy.settings). The web dashboard falls back to
   147→    // BrowserOnlyEngine.
   148→
   149→    this.running = true;
   150→    try {
   151→      // Step 1: tell the companion to use this config
   152→      const ctrlResp = await fetch("http://127.0.0.1:10809/api/tunnel/start", {
   153→        method: "POST",
   154→        headers: { "Content-Type": "application/json" },
   155→        body: JSON.stringify({ configUri }),
   156→        signal: AbortSignal.timeout(timeoutMs),
   157→      });
   158→      if (!ctrlResp.ok) {
   159→        return {
   160→          started: false,
   161→          trafficVerified: false,
   162→          exitIp: null,
   163→          realIp: null,
   164→          error: `Companion control API returned ${ctrlResp.status}`,
   165→          latencyMs: null,
   166→        };
   167→      }
   168→      const ctrlData = await ctrlResp.json() as { ok: boolean; error?: string };
   169→
   170→      if (!ctrlData.ok) {
   171→        return {
   172→          started: false,
   173→          trafficVerified: false,
   174→          exitIp: null,
   175→          realIp: null,
   176→          error: ctrlData.error || "Companion failed to start tunnel",
   177→          latencyMs: null,
   178→        };
   179→      }
   180→
   181→      // Step 2: the companion has started xray. Now we need to verify
   182→      // traffic flows through it. But we can't use SOCKS5 from fetch()
   183→      // directly in a web page. So we ask the companion to do the
   184→      // verification for us and report back.
   185→      const verifyResp = await fetch("http://127.0.0.1:10809/api/tunnel/verify", {
   186→        method: "POST",
   187→        signal: AbortSignal.timeout(timeoutMs),
   188→      });
   189→      const verifyData = await verifyResp.json() as {
   190→        trafficVerified: boolean;
   191→        exitIp: string | null;
   192→        realIp: string | null;
   193→        latencyMs: number | null;
   194→        error?: string;
   195→      };
   196→
   197→      return {
   198→        started: true,
   199→        trafficVerified: verifyData.trafficVerified,
   200→        exitIp: verifyData.exitIp,
   201→        realIp: verifyData.realIp,
   202→        error: verifyData.error || null,
   203→        latencyMs: verifyData.latencyMs,
   204→      };
   205→    } catch (e) {
   206→      const err = e as Error;
   207→      return {
   208→        started: false,
   209→        trafficVerified: false,
   210→        exitIp: null,
   211→        realIp: null,
   212→        error: `Companion unreachable: ${err.message}. Is the Distort Desktop app running?`,
   213→        latencyMs: null,
   214→      };
   215→    } finally {
   216→      this.running = false;
   217→    }
   218→  }
   219→
   220→  async disconnect(): Promise<void> {
   221→    try {
   222→      await fetch("http://127.0.0.1:10809/api/tunnel/stop", { method: "POST" });
   223→    } catch {
   224→      // Best-effort
   225→    }
   226→    this.running = false;
   227→  }
   228→
   229→  isRunning(): boolean {
   230→    return this.running;
   231→  }
   232→}

   234→// Detect whether the desktop companion is running.
   235→// Tries to hit the companion's health endpoint.
   236→export async function detectCompanion(timeoutMs = 2000): Promise<boolean> {
   237→  try {
   238→    const resp = await fetch("http://127.0.0.1:10809/api/health", {
   239→      signal: AbortSignal.timeout(timeoutMs),
   240→    });
   241→    return resp.ok;
   242→  } catch {
   243→    return false;
   244→  }
   245→}

   247→// Factory: pick the right engine for the current context.
   248→export async function createTunnelEngine(): Promise<TunnelEngine> {
   249→  // Only run this in the browser
   250→  if (typeof window === "undefined") {
   251→    // Server-side rendering — return a no-op engine
   252→    return new BrowserOnlyEngine();
   253→  }
   254→  const companionUp = await detectCompanion();
   255→  return companionUp
   256→    ? new BrowserWithCompanionEngine()
   257→    : new BrowserOnlyEngine();
   258→}

   260→// ---------------------------------------------------------------------------
   261→// The recovery engine itself
   262→// ---------------------------------------------------------------------------
   263→
   264→export type RecoveryEventCallback = (status: RecoveryStatus) => void;
   265→
   266→export type RecoveryEngineOptions = {
   267→  // The tunnel engine to use (auto-detected if not provided)
   268→  engine?: TunnelEngine;
   269→  // Whether to fetch remote candidates from the API (default true)
   270→  fetchRemoteCandidates?: boolean;
   271→  // Max attempts override (default from policy)
   272→  maxAttempts?: number;
   273→  // Callback for status updates (for UI binding)
   274→  onStatusChange?: RecoveryEventCallback;
   275→};
   276→
   277→export class RecoveryEngine {
   278→  private status: RecoveryStatus;
   279→  private engine: TunnelEngine | null = null;
   280→  private onStatusChange: RecoveryEventCallback | null;
   281→  private abortController: AbortController | null = null;
   282→  private monitoringInterval: ReturnType<typeof setInterval> | null = null;
   283→  private networkChangeCleanup: (() => void) | null = null;
   284→  private fetchRemote: boolean;
   285→
   286→  constructor(opts: RecoveryEngineOptions = {}) {
   287→    this.status = idleStatus();
   288→    this.engine = opts.engine || null;
   289→    this.onStatusChange = opts.onStatusChange || null;
   290→    this.fetchRemote = opts.fetchRemoteCandidates ?? true;
   291→  }
   292→
   293→  // Get current status (without triggering updates)
   294→  getStatus(): RecoveryStatus {
   295→    return { ...this.status };
   296→  }
   297→
   298→  // Update status + notify listeners
   299→  private setStatus(newStatus: RecoveryStatus): void {
   300→    this.status = newStatus;
   301→    this.onStatusChange?.(newStatus);
   302→  }
   303→
   304→  private emit(event: Parameters<typeof transition>[1]): void {
   305→    const newStatus = transition(this.status, event);
   306→    if (newStatus !== this.status) {
   307→      this.setStatus(newStatus);
   308→    }
   309→  }
   310→
   311→  // Add an attempt to the log + report to server (best-effort)
   312→  private logAttempt(attempt: RecoveryAttempt): void {
   313→    this.status = {
   314→      ...this.status,
   315→      attempts: [attempt, ...this.status.attempts].slice(0, 50),
   316→    };
   317→    this.setStatus(this.status);
   318→
   319→    // Best-effort report to server (don't block on it)
   320→    if (typeof fetch !== "undefined") {
   321→      fetch("/api/emergency/report", {
   322→        method: "POST",
   323→        headers: { "Content-Type": "application/json" },
   324→        body: JSON.stringify({
   325→          planId: attempt.planId,
   326→          attempt: {
   327→            candidateId: attempt.candidateId,
   328→            candidateLabel: attempt.candidateLabel,
   329→            attemptNumber: attempt.attemptNumber,
   330→            state: attempt.state,
   331→            result: attempt.result,
   332→            reason: attempt.reason,
   333→            latencyMs: attempt.latencyMs,
   334→            startedAt: attempt.startedAt,
   335→            endedAt: attempt.endedAt,
   336→          },
   337→        }),
   338→      }).catch(() => {
   339→        // Network may be down — that's OK, we keep the local log
   340→      });
   341→    }
   342→  }
   343→
   344→  // The main recovery cycle. Returns the final status.
   345→  async recover(): Promise<RecoveryStatus> {
   346→    // Set up abort signal
   347→    this.abortController = new AbortController();
   348→
   349→    // 1. Start → diagnosing
   350→    this.emit({ type: "start" });
   351→
   352→    // 2. Diagnose the network (real client-side probes)
   353→    const diagnosis = await diagnoseClientSide();
   354→
   355→    // 3. If no base network, fail closed immediately
   356→    if (diagnosis.baseNetwork === "offline") {
   357→      this.emit({
   358→        type: "no-base-network",
   359→        reason: "Device is offline (navigator.onLine=false). No network interface is active. Connect to Wi-Fi or mobile data, then retry.",
   360→      });
   361→      this.logAttempt({
   362→        planId: "no-plan",
   363→        candidateId: "none",
   364→        candidateLabel: "Network diagnosis",
   365→        attemptNumber: 1,
   366→        state: "exhausted",
   367→        result: "failure",
   368→        reason: diagnosis.summary,
   369→        latencyMs: null,
   370→        startedAt: new Date().toISOString(),
   371→        endedAt: new Date().toISOString(),
   372→      });
   373→      return this.getStatus();
   374→    }
   375→
   376→    // 4. Gather candidates: local configs first, then remote (if fetchable)
   377→    const candidates = await this.gatherCandidates(diagnosis);
   378→
   379→    if (candidates.length === 0) {
   380→      this.emit({
   381→        type: "no-candidates-left",
   382→        reason: "No VPN configs available. Import a config from the Community Mesh tab or paste your own. Without at least one config, the recovery engine has nothing to try.",
   383→      });
   384→      this.logAttempt({
   385→        planId: "no-plan",
   386→        candidateId: "none",
   387→        candidateLabel: "Candidate gathering",
   388→        attemptNumber: 1,
   389→        state: "exhausted",
   390→        result: "failure",
   391→        reason: "No candidates available locally or remotely.",
   392→        latencyMs: null,
   393→        startedAt: new Date().toISOString(),
   394→        endedAt: new Date().toISOString(),
   395→      });
   396→      return this.getStatus();
   397→    }
   398→
   399→    // 5. Build the plan (client-side — we rank locally)
   400→    const plan: RecoveryPlan = {
   401→      planId: `plan_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   402→      diagnosis,
   403→      candidates,
   404→      policy: {
   405→        maxAttempts: 10,
   406→        verifyTimeoutMs: 8000,
   407→        baseDelayMs: 1500,
   408→        maxDelayMs: 30000,
   409→        maxConsecutiveFailures: 5,
   410→        autoRetryAfterExhaustion: false,
   411→        enforceKillSwitch: true,
   412→        requireDnsLeakProtection: true,
   413→      },
   414→      rationale: `Client-side plan: ${candidates.length} candidates. Base network: ${diagnosis.baseNetwork}.`,
   415→      confidence: candidates.length >= 5 ? "medium" : "low",
   416→      createdAt: new Date().toISOString(),
   417→      userTier: "free", // determined client-side; server verifies
   418→    };
   419→
   420→    this.status = { ...this.status, activePlan: plan, attemptsThisCycle: 0 };
   421→    this.setStatus(this.status);
   422→
   423→    // 6. Diagnosis complete → selecting
   424→    this.emit({ type: "diagnosis-complete" });
   425→
   426→    // 7. Try each candidate in sequence
   427→    for (let i = 0; i < plan.candidates.length; i++) {
   428→      if (this.abortController?.signal.aborted) break;
   429→
   430→      const candidate = plan.candidates[i];
   431→      const attemptNumber = i + 1;
   432→
   433→      // Select this candidate
   434→      this.emit({ type: "candidate-selected", candidate });
   435→
   436→      // connecting
   437→      const attemptStart = new Date().toISOString();
   438→
   439→      // If the engine can't connect, fail honestly
   440→      if (!this.engine || !this.engine.canConnect) {
   441→        const reason = this.engine
   442→          ? `Engine "${this.engine.platform}" cannot establish tunnels in this context.`
   443→          : "No tunnel engine available.";
   444→
   445→        this.emit({ type: "engine-failed", reason });
   446→
   447→        this.logAttempt({
   448→          planId: plan.planId,
   449→          candidateId: candidate.id,
   450→          candidateLabel: candidate.label,
   451→          attemptNumber,
   452→          state: "recovering",
   453→          result: "failure",
   454→          reason,
   455→          latencyMs: null,
   456→          startedAt: attemptStart,
   457→          endedAt: new Date().toISOString(),
   458→        });
   459→
   460→        await markFailed(candidate.id);
   461→
   462→        // Try next candidate
   463→        const next = plan.candidates[i + 1];
   464→        if (next) {
   465→          // Backoff before next try
   466→          await this.delay(1500);
   467→          this.emit({ type: "next-candidate-available", candidate: next });
   468→          continue;
   469→        } else {
   470→          this.emit({
   471→            type: "no-candidates-left",
   472→            reason: `All ${plan.candidates.length} candidates tried. Engine "${this.engine?.platform}" cannot establish tunnels. Install the Distort Desktop or Android app for real connectivity.`,
   473→          });
   474→          break;
   475→        }
   476→      }
   477→
   478→      // Engine CAN connect — try it
   479→      // First, fetch the configUri on-demand (not pre-fetched in the plan)
   480→      let configUri = candidate.configUri;
   481→      if (!configUri && candidate.id !== "satellite-stub-vip") {
   482→        // Remote candidate — fetch URI from secure endpoint
   483→        configUri = await this.fetchCandidateUri(candidate.id);
   484→        if (!configUri) {
   485→          // Could not fetch URI (rate limited, auth failed, or not found)
   486→          // Skip this candidate — don't claim "connected"
   487→          this.emit({
   488→            type: "verification-failed",
   489→            reason: "Could not fetch config URI for this candidate (rate limited, auth required, or not found).",
   490→          });
   491→
   492→          this.logAttempt({
   493→            planId: plan.planId,
   494→            candidateId: candidate.id,
   495→            candidateLabel: candidate.label,
   496→            attemptNumber,
   497→            state: "recovering",
   498→            result: "failure",
   499→            reason: "Config URI fetch failed — skipped.",
   500→            latencyMs: null,
   501→            startedAt: attemptStart,
   502→            endedAt: new Date().toISOString(),
   503→          });
   504→
   505→          await markFailed(candidate.id);
   506→
   507→          const next = plan.candidates[i + 1];
   508→          if (next) {
   509→            await this.delay(1500);
   510→            this.emit({ type: "next-candidate-available", candidate: next });
   511→            continue;
   512→          } else {
   513→            this.emit({
   514→              type: "no-candidates-left",
   515→              reason: `All ${plan.candidates.length} candidates tried. Could not fetch URI for remote candidates (rate limited or auth required).`,
   516→            });
   517→            break;
   518→          }
   519→        }
   520→      }
   521→
   522→      const result = await this.engine.connect(configUri || "", plan.policy.verifyTimeoutMs);
   523→
   524→      if (result.started && result.trafficVerified && result.exitIp !== result.realIp) {
   525→        // SUCCESS — tunnel verified, traffic flows, exit IP differs from real IP
   526→        this.emit({
   527→          type: "traffic-verified",
   528→          exitIp: result.exitIp,
   529→          exitCountry: null, // could do a lookup later
   530→        });
   531→
   532→        await markVerified(candidate.id);
   533→
   534→        this.logAttempt({
   535→          planId: plan.planId,
   536→          candidateId: candidate.id,
   537→          candidateLabel: candidate.label,
   538→          attemptNumber,
   539→          state: "connected",
   540→          result: "success",
   541→          reason: `Tunnel verified. Exit IP: ${result.exitIp}. Latency: ${result.latencyMs}ms.`,
   542→          latencyMs: result.latencyMs,
   543→          startedAt: attemptStart,
   544→          endedAt: new Date().toISOString(),
   545→        });
   546→
   547→        // Start health monitoring
   548→        this.startHealthMonitoring();
   549→        return this.getStatus();
   550→      }
   551→
   552→      // FAILURE — either engine didn't start, traffic didn't verify, or leak detected
   553→      let reason: string;
   554→      if (!result.started) {
   555→        reason = result.error || "Engine failed to start.";
   556→      } else if (result.exitIp === result.realIp && result.exitIp !== null) {
   557→        reason = `Traffic leak detected: exit IP (${result.exitIp}) equals real IP. Tunnel is not actually routing traffic.`;
   558→      } else if (!result.trafficVerified) {
   559→        reason = result.error || "Tunnel started but HTTP verification failed — server may be unreachable or credentials invalid.";
   560→      } else {
   561→        reason = result.error || "Unknown failure.";
   562→      }
   563→
   564→      this.emit({ type: "verification-failed", reason });
   565→
   566→      await markFailed(candidate.id);
   567→
   568→      this.logAttempt({
   569→        planId: plan.planId,
   570→        candidateId: candidate.id,
   571→        candidateLabel: candidate.label,
   572→        attemptNumber,
   573→        state: "recovering",
   574→        result: "failure",
   575→        reason,
   576→        latencyMs: result.latencyMs,
   577→        startedAt: attemptStart,
   578→        endedAt: new Date().toISOString(),
   579→      });
   580→
   581→      // Check exhaustion
   582→      this.status = {
   583→        ...this.status,
   584→        attemptsThisCycle: attemptNumber,
   585→      };
   586→      if (attemptNumber >= plan.policy.maxAttempts) {
   587→        this.emit({
   588→          type: "no-candidates-left",
   589→          reason: `Max attempts (${plan.policy.maxAttempts}) reached. ${attemptNumber} candidates tried, none verified. Last failure: ${reason}`,
   590→        });
   591→        break;
   592→      }
   593→
   594→      // Try next candidate
   595→      const next = plan.candidates[i + 1];
   596→      if (next) {
   597→        // Exponential backoff
   598→        const delayMs = Math.min(
   599→          plan.policy.baseDelayMs * Math.pow(2, attemptNumber - 1),
   600→          plan.policy.maxDelayMs
   601→        );
   602→        await this.delay(delayMs);
   603→        this.emit({ type: "next-candidate-available", candidate: next });
   604→      } else {
   605→        this.emit({
   606→          type: "no-candidates-left",
   607→          reason: `All ${plan.candidates.length} candidates exhausted. None verified tunnel traffic. Install the Distort Desktop or Android app for real connectivity, or import fresh configs from the Community Mesh.`,
   608→        });
   609→        break;
   610→      }
   611→    }
   612→
   613→    return this.getStatus();
   614→  }
   615→
   616→  // Gather candidates: local configs first (offline-available), then remote.
   617→  private async gatherCandidates(diagnosis: NetworkDiagnosis): Promise<CandidatePath[]> {
   618→    const candidates: CandidatePath[] = [];
   619→
   620→    // 1. Local configs (from IndexedDB — available offline)
   621→    try {
   622→      const local = await getAllConfigs();
   623→      // Sort: verified first, then by score
   624→      const sorted = [...local].sort((a, b) => {
   625→        if (a.lastVerifiedAt && !b.lastVerifiedAt) return -1;
   626→        if (!a.lastVerifiedAt && b.lastVerifiedAt) return 1;
   627→        return b.score - a.score;
   628→      });
   629→      for (const cfg of sorted) {
   630→        candidates.push(this.storedToCandidate(cfg));
   631→      }
   632→    } catch {
   633→      // IndexedDB not available — skip local configs
   634→    }
   635→
   636→    // 2. Remote configs (from API — only if internet is up)
   637→    // SECURITY: We fetch the PLAN (metadata only, no configUri) from
   638→    // /api/emergency/plan. The configUri is fetched separately per-candidate
   639→    // via POST /api/emergency/candidates/[id]/uri, which is rate-limited and
   640→    // (in production) authenticated. This prevents mass credential harvesting.
   641→    if (this.fetchRemote && diagnosis.internationalReachable) {
   642→      try {
   643→        const resp = await fetch("/api/emergency/plan", {
   644→          method: "POST",
   645→          headers: { "Content-Type": "application/json" },
   646→          body: JSON.stringify({ userTier: "free" }),
   647→        });
   648→        if (resp.ok) {
   649→          const data = await resp.json() as RecoveryPlan;
   650→          for (const cfg of data.candidates) {
   651→            // Don't duplicate configs we already have locally
   652→            if (candidates.some((c) => c.id === cfg.id)) continue;
   653→            candidates.push({
   654→              ...cfg,
   655→              // configUri is empty in the plan response — will be fetched
   656→              // on-demand when this candidate is actually tried
   657→              configUri: "",
   658→            });
   659→          }
   660→        }
   661→      } catch {
   662→        // API unreachable — fall back to local only
   663→      }
   664→    }
   665→
   666→    return candidates;
   667→  }
   668→
   669→  // Fetch the configUri for a candidate from the secure endpoint.
   670→  // Called on-demand right before the engine tries to connect — never pre-fetched.
   671→  // Returns null if the fetch fails (candidate is skipped).
   672→  private async fetchCandidateUri(candidateId: string): Promise<string | null> {
   673→    try {
   674→      const resp = await fetch(`/api/emergency/candidates/${candidateId}/uri`, {
   675→        method: "POST",
   676→        // credentials: 'same-origin' is the default — session cookie (if any) is sent
   677→      });
   678→      if (!resp.ok) return null;
   679→      const data = await resp.json() as { uri?: string };
   680→      return data.uri || null;
   681→    } catch {
   682→      return null;
   683→    }
   684→  }
   685→
   686→  private storedToCandidate(cfg: StoredConfig): CandidatePath {
   687→    const c = toCandidate(cfg);
   688→    return {
   689→      ...c,
   690→      protocol: c.protocol as TransportType,
   691→      lastProbe: cfg.lastVerifiedAt
   692→        ? {
   693→            configId: cfg.id,
   694→            structural: "valid",
   695→            reachability: "reachable",
   696→            tunnel: "verified",
   697→            latencyMs: null,
   698→            exitIp: null,
   699→            exitCountry: cfg.country,
   700→            probedAt: cfg.lastVerifiedAt,
   701→            failureReason: null,
   702→          }
   703→        : null,
   704→    };
   705→  }

   707→  // Health monitoring — runs every 30s after connected.
   708→  // Detects quality degradation and triggers recovery if needed.
   709→  private startHealthMonitoring(): void {
   710→    this.monitoringInterval = setInterval(async () => {
   711→      if (this.status.state !== "connected" && this.status.state !== "degraded") {
   712→        this.stopHealthMonitoring();
   713→        return;
   714→      }
   715→      const health = await quickHealthCheck(3000);
   716→      if (health.unknown) {
   717→        // Ambiguous result — don't trigger recovery, just log.
   718→        // Recovery on "unknown" would cause false-positive disconnects.
   719→        return;
   720→      }
   721→      if (!health.online) {
   722→        // Network dropped — trigger recovery
   723→        this.emit({ type: "quality-dropped" });
   724→        // Auto-recover after a short delay
   725→        setTimeout(() => {
   726→          if (this.status.state === "degraded") {
   727→            this.recover().catch(() => {});
   728→          }
   729→        }, 2000);
   730→      } else if (health.latencyMs !== null && health.latencyMs > 2000) {
   731→        // High latency — degraded
   732→        if (this.status.state === "connected") {
   733→          this.emit({ type: "quality-dropped" });
   734→        }
   735→      } else {
   736→        // Healthy — if we were degraded, recover
   737→        if (this.status.state === "degraded") {
   738→          this.emit({ type: "quality-recovered" });
   739→        }
   740→      }
   741→    }, 30000);
   742→
   743→    // Listen for network change events (Wi-Fi → mobile data switch)
   744→    this.networkChangeCleanup = onNetworkChange(() => {
   745→      // Re-diagnose on network change
   746→      if (this.status.state === "connected") {
   747→        this.emit({ type: "quality-dropped" });
   748→        setTimeout(() => {
   749→          if (this.status.state === "degraded") {
   750→            this.recover().catch(() => {});
   751→          }
   752→        }, 1000);
   753→      }
   754→    });
   755→  }
   756→
   757→  private stopHealthMonitoring(): void {
   758→    if (this.monitoringInterval) {
   759→      clearInterval(this.monitoringInterval);
   760→      this.monitoringInterval = null;
   761→    }
   762→    if (this.networkChangeCleanup) {
   763→      this.networkChangeCleanup();
   764→      this.networkChangeCleanup = null;
   765→    }
   766→  }
   767→
   768→  private delay(ms: number): Promise<void> {
   769→    return new Promise((resolve) => {
   770→      setTimeout(resolve, Math.min(ms, 30000));
   771→    });
   772→  }
   773→
   774→  // Manual disconnect
   775→  async disconnect(): Promise<void> {
   776→    this.stopHealthMonitoring();
   777→    if (this.engine) {
   778→      await this.engine.disconnect();
   779→    }
   780→    this.emit({ type: "disconnect" });
   781→  }
   782→
   783→  // Abort the current recovery cycle
   784→  abort(): void {
   785→    this.abortController?.abort();
   786→    this.stopHealthMonitoring();
   787→    this.emit({ type: "abort" });
   788→  }
   789→
   790→  // Reset to idle
   791→  reset(): void {
   792→    this.stopHealthMonitoring();
   793→    this.abortController?.abort();
   794→    this.emit({ type: "reset" });
   795→  }

   797→  // Initialize the engine (detect companion, etc.)
   798→  async init(): Promise<void> {
   799→    if (!this.engine) {
   800→      this.engine = await createTunnelEngine();
   801→    }
   802→  }
