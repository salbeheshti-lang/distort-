    59→function flagFor(country?: string | null): string {
    60→  if (!country) return "🌐";
    61→  const map: Record<string, string> = {
    62→    Iran: "🇮🇷", Germany: "🇩🇪", Netherlands: "🇳🇱", France: "🇫🇷", Finland: "🇫🇮",
    63→    "United States": "🇺🇸", "United Kingdom": "🇬🇧", Japan: "🇯🇵", Singapore: "🇸🇬",
    64→    Turkey: "🇹🇷", UAE: "🇦🇪", Georgia: "🇬🇪", Armenia: "🇦🇲",
    65→  };
    66→  return map[country] || "🌐";
    67→}

    69→function reachColor(reach: string): string {
    70→  if (reach === "ok") return "#10d9a0";
    71→  if (reach === "fail") return "#ff4d6d";
    72→  return "#8aa0c4";
    73→}
    74→
    75→function scoreColor(score: number): string {
    76→  if (score >= 80) return "#10d9a0";
    77→  if (score >= 60) return "#00f0ff";
    78→  if (score >= 40) return "#ffb020";
    79→  return "#8aa0c4";
    80→}

   408→function ImportDialog({ open, onOpenChange, onDone }: { open: boolean; onOpenChange: (o: boolean) => void; onDone: () => void }) {
   409→  const [raw, setRaw] = useState("");
   410→  const [source, setSource] = useState("personal");
   411→  const [busy, setBusy] = useState(false);
   412→
   413→  const handleImport = async () => {
   414→    if (!raw.trim()) return;
   415→    setBusy(true);
   416→    try {
   417→      const r = await postJSON<{ count: number; config?: unknown }>("/api/imported-configs/import", { raw, source });
   418→      toast.success("Configs imported", { description: `${r.count} parsed & saved.` });
   419→      setRaw("");
   420→      onOpenChange(false);
   421→      onDone();
   422→    } catch (e) {
   423→      toast.error("Import failed", { description: (e as Error).message });
   424→    } finally {
   425→      setBusy(false);
   426→    }
   427→  };
   428→
   429→  return (
   430→    <Dialog open={open} onOpenChange={onOpenChange}>
   431→      <DialogContent className="glass-strong border-[#00f0ff]/20 max-w-2xl">

   470→function AddSubDialog({ open, onOpenChange, onDone }: { open: boolean; onOpenChange: (o: boolean) => void; onDone: () => void }) {
   471→  const [url, setUrl] = useState("");
   472→  const [label, setLabel] = useState("");
   473→  const [interval, setInterval] = useState(60);
   474→  const [busy, setBusy] = useState(false);
   475→
   476→  const handleAdd = async () => {
   477→    if (!url.trim()) return;
   478→    setBusy(true);
   479→    try {
   480→      const r = await postJSON<{ subscription: Subscription; warning?: string }>("/api/subscriptions", {
   481→        url, label: label || undefined, kind: "custom", intervalMin: interval, fetchNow: true,
   482→      });
   483→      if (r.warning) {
   484→        toast.warning("Subscription added (fetch warning)", { description: r.warning });
   485→      } else {
   486→        toast.success("Subscription added", { description: `${r.subscription.label}: ${r.subscription.lastCount} configs` });
   487→      }
   488→      setUrl(""); setLabel("");
   489→      onOpenChange(false);
   490→      onDone();
   491→    } catch (e) {
   492→      toast.error("Could not add subscription", { description: (e as Error).message });
   493→    } finally {
   494→      setBusy(false);
   495→    }
   496→  };
   497→
   498→  return (
   499→    <Dialog open={open} onOpenChange={onOpenChange}>
   500→      <DialogContent className="glass-strong border-[#00f0ff]/20 max-w-lg">
