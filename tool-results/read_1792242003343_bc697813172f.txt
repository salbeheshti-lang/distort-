    59→    59→function flagFor(country?: string | null): string {
    60→    60→  if (!country) return "🌐";
    61→    61→  const map: Record<string, string> = {
    62→    62→    Iran: "🇮🇷", Germany: "🇩🇪", Netherlands: "🇳🇱", France: "🇫🇷", Finland: "🇫🇮",
    63→    63→    "United States": "🇺🇸", "United Kingdom": "🇬🇧", Japan: "🇯🇵", Singapore: "🇸🇬",
    64→    64→    Turkey: "🇹🇷", UAE: "🇦🇪", Georgia: "🇬🇪", Armenia: "🇦🇲",
    65→    65→  };
    66→    66→  return map[country] || "🌐";
    67→    67→}

    69→    69→function reachColor(reach: string): string {
    70→    70→  if (reach === "ok") return "#10d9a0";
    71→    71→  if (reach === "fail") return "#ff4d6d";
    72→    72→  return "#8aa0c4";
    73→    73→}
    74→    74→
    75→    75→function scoreColor(score: number): string {
    76→    76→  if (score >= 80) return "#10d9a0";

    78→    78→  if (score >= 40) return "#ffb020";
    79→    79→  return "#8aa0c4";
    80→    80→}
