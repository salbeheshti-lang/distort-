     1→     1→     1→// DISTORT — Client-side recovery engine
     2→     2→     2→//
     3→     3→     3→// THE ACTUAL RECOVERY LOOP. Runs entirely in the browser/app.
     4→     4→     4→//
     5→     5→     5→// Per the user's design (§3):
     6→     6→     6→//   Start → Diagnose network → Gather options (local + remote) →
     7→     7→     7→//   Validate + prioritize → Connect with real engine → Verify →
     8→     8→     8→//   (success: monitor health) | (fail: try next or report no path)
     9→     9→     9→//
    10→    10→    10→// CRITICAL HONESTY RULES:
    11→    11→    11→//   1. "connected" is ONLY set after a real HTTP request succeeds through
    12→    12→    12→//      the tunnel AND the exit IP differs from the real IP.
    13→    13→    13→//   2. If no path works, state goes to "exhausted" with a clear reason —
    14→    14→    14→//      never a fake "connected".
    15→    15→    15→//   3. The server's plan is a SUGGESTION — the client does its own probing.
    16→    16→    16→//   4. No credentials are ever logged (use scrubSensitive on any string
    17→    17→    17→//      before displaying or persisting).
    18→    18→    18→//
    19→    19→    19→// BROWSER LIMITATIONS (documented honestly):
    20→    20→    20→//   - Cannot spawn xray-core (needs native process)
    21→    21→    21→//   - Cannot open raw TCP sockets
    22→    22→    22→//   - Cannot set system proxy (only chrome.proxy for the extension)
    23→    23→    23→//   - CAN: diagnose network, validate config structure, fetch plan from API,
    24→    24→    24→//     and — IF the desktop companion is running — delegate the actual
    25→    25→    25→//     tunnel to it via the local SOCKS5 proxy at 127.0.0.1:10808.
    26→    26→    26→//
    27→    27→    27→// DESKTOP COMPANION BRIDGE:
    28→    28→    28→//   When the Distort Desktop app is running, it exposes a local SOCKS5
    29→    29→    29→//   proxy on 127.0.0.1:10808. The browser can route fetch requests through
    30→    30→    30→//   it (via a PAC script or chrome.proxy.settings). If we detect the
    31→    31→    31→//   companion is up, we can do real tunnel verification from the browser
    32→    32→    32→//   by fetching through the SOCKS5 proxy and comparing the exit IP.
    33→    33→    33→//
    34→    34→    34→// WITHOUT DESKTOP COMPANION:
    35→    35→    35→//   The browser cannot actually establish a VPN tunnel. In this case, the
    36→    36→    36→//   engine diagnoses the network, lists available configs, but reports
    37→    37→    37→//   honestly: "Install the desktop or Android app to actually connect.
    38→    38→    38→//   The web dashboard can only diagnose and validate configs."

   120→   120→   120→// ---------------------------------------------------------------------------
   121→   121→   121→
   122→   122→   122→export class BrowserWithCompanionEngine implements TunnelEngine {
   123→   123→   123→  readonly platform = "browser-with-companion";
   124→   124→   124→  readonly canConnect = true;
   125→   125→   125→  private running = false;
   126→   126→   126→
   127→   127→   127→  async connect(configUri: string, timeoutMs: number): Promise<TunnelEngineResult> {
   128→   128→   128→    // The companion app (Distort Desktop) is running and exposes:
   129→   129→   129→    //   - SOCKS5 at 127.0.0.1:10808
   130→   130→   130→    //   - A control API at 127.0.0.1:10809 (HTTP)
   131→   131→   131→    //
   132→   132→   132→    // Steps:
   133→   133→   133→    //   1. POST the configUri to the companion's control API
   134→   134→   134→    //      (it spawns xray with that config)
   135→   135→   135→    //   2. Wait for the companion to confirm xray started
   136→   136→   136→    //   3. Fetch exit IP through the SOCKS5 proxy
   137→   137→   137→    //   4. Fetch real IP directly (bypassing proxy)
   138→   138→   138→    //   5. Compare — if different, tunnel is verified
   139→   139→   139→    //
   140→   140→   140→    // BROWSER LIMITATION: fetch() cannot use a SOCKS5 proxy directly.
   141→   141→   141→    // We'd need chrome.proxy.settings (extension only) or a custom
   142→   142→   142→    // SOCKS5-over-WebSocket bridge. For the extension, this works.
   143→   143→   143→    // For the plain web app, we can only do the control API call.
   144→   144→   144→    //
   145→   145→   145→    // So in practice, this engine works ONLY in the Chrome extension context
   146→   146→   146→    // (which has chrome.proxy.settings). The web dashboard falls back to
   147→   147→   147→    // BrowserOnlyEngine.
   148→   148→   148→
   149→   149→   149→    this.running = true;
   150→   150→   150→    try {
   151→   151→   151→      // Step 1: tell the companion to use this config
   152→   152→   152→      const ctrlResp = await fetch("http://127.0.0.1:10809/api/tunnel/start", {
   153→   153→   153→        method: "POST",
   154→   154→   154→        headers: { "Content-Type": "application/json" },
   155→   155→   155→        body: JSON.stringify({ configUri }),
   156→   156→   156→        signal: AbortSignal.timeout(timeoutMs),
   157→   157→   157→      });
   158→   158→   158→      if (!ctrlResp.ok) {
   159→   159→   159→        return {
   160→   160→   160→          started: false,
   161→   161→   161→          trafficVerified: false,
   162→   162→   162→          exitIp: null,
   163→   163→   163→          realIp: null,
   164→   164→   164→          error: `Companion control API returned ${ctrlResp.status}`,
   165→   165→   165→          latencyMs: null,
   166→   166→   166→        };
   167→   167→   167→      }
   168→   168→   168→      const ctrlData = await ctrlResp.json() as { ok: boolean; error?: string };
   169→   169→   169→
   170→   170→   170→      if (!ctrlData.ok) {
   171→   171→   171→        return {
   172→   172→   172→          started: false,
   173→   173→   173→          trafficVerified: false,
   174→   174→   174→          exitIp: null,
   175→   175→   175→          realIp: null,
   176→   176→   176→          error: ctrlData.error || "Companion failed to start tunnel",
   177→   177→   177→          latencyMs: null,
   178→   178→   178→        };
   179→   179→   179→      }
   180→   180→   180→
   181→   181→   181→      // Step 2: the companion has started xray. Now we need to verify
   182→   182→   182→      // traffic flows through it. But we can't use SOCKS5 from fetch()
   183→   183→   183→      // directly in a web page. So we ask the companion to do the
   184→   184→   184→      // verification for us and report back.
   185→   185→   185→      const verifyResp = await fetch("http://127.0.0.1:10809/api/tunnel/verify", {
   186→   186→   186→        method: "POST",
   187→   187→   187→        signal: AbortSignal.timeout(timeoutMs),
   188→   188→   188→      });
   189→   189→   189→      const verifyData = await verifyResp.json() as {
   190→   190→   190→        trafficVerified: boolean;
   191→   191→   191→        exitIp: string | null;
   192→   192→   192→        realIp: string | null;
   193→   193→   193→        latencyMs: number | null;
   194→   194→   194→        error?: string;
   195→   195→   195→      };
   196→   196→   196→
   197→   197→   197→      return {
   198→   198→   198→        started: true,
   199→   199→   199→        trafficVerified: verifyData.trafficVerified,
   200→   200→   200→        exitIp: verifyData.exitIp,
   201→   201→   201→        realIp: verifyData.realIp,
   202→   202→   202→        error: verifyData.error || null,
   203→   203→   203→        latencyMs: verifyData.latencyMs,
   204→   204→   204→      };
   205→   205→   205→    } catch (e) {
   206→   206→   206→      const err = e as Error;
   207→   207→   207→      return {
   208→   208→   208→        started: false,
   209→   209→   209→        trafficVerified: false,
   210→   210→   210→        exitIp: null,
   211→   211→   211→        realIp: null,
   212→   212→   212→        error: `Companion unreachable: ${err.message}. Is the Distort Desktop app running?`,
   213→   213→   213→        latencyMs: null,
   214→   214→   214→      };
   215→   215→   215→    } finally {
   216→   216→   216→      this.running = false;
   217→   217→   217→    }
   218→   218→   218→  }
   219→   219→   219→

   223→   223→   223→    } catch {
   224→   224→   224→      // Best-effort
   225→   225→   225→    }
   226→   226→   226→    this.running = false;
   227→   227→   227→  }
   228→   228→   228→
   229→   229→   229→  isRunning(): boolean {
   230→   230→   230→    return this.running;
   231→   231→   231→  }

   277→   277→   277→export class RecoveryEngine {
   278→   278→   278→  private status: RecoveryStatus;
   279→   279→   279→  private engine: TunnelEngine | null = null;
   280→   280→   280→  private onStatusChange: RecoveryEventCallback | null;
   281→   281→   281→  private abortController: AbortController | null = null;
   282→   282→   282→  private monitoringInterval: ReturnType<typeof setInterval> | null = null;
   283→   283→   283→  private networkChangeCleanup: (() => void) | null = null;
   284→   284→   284→  private fetchRemote: boolean;
   285→   285→   285→
   286→   286→   286→  constructor(opts: RecoveryEngineOptions = {}) {
   287→   287→   287→    this.status = idleStatus();
   288→   288→   288→    this.engine = opts.engine || null;
   289→   289→   289→    this.onStatusChange = opts.onStatusChange || null;
   290→   290→   290→    this.fetchRemote = opts.fetchRemoteCandidates ?? true;
   291→   291→   291→  }
   292→   292→   292→
   293→   293→   293→  // Get current status (without triggering updates)
   294→   294→   294→  getStatus(): RecoveryStatus {
   295→   295→   295→    return { ...this.status };
   296→   296→   296→  }
   297→   297→   297→
   298→   298→   298→  // Update status + notify listeners
   299→   299→   299→  private setStatus(newStatus: RecoveryStatus): void {
   300→   300→   300→    this.status = newStatus;
   301→   301→   301→    this.onStatusChange?.(newStatus);
   302→   302→   302→  }
   303→   303→   303→
   304→   304→   304→  private emit(event: Parameters<typeof transition>[1]): void {
   305→   305→   305→    const newStatus = transition(this.status, event);
   306→   306→   306→    if (newStatus !== this.status) {
   307→   307→   307→      this.setStatus(newStatus);
   308→   308→   308→    }
   309→   309→   309→  }
   310→   310→   310→
   311→   311→   311→  // Add an attempt to the log + report to server (best-effort)
   312→   312→   312→  private logAttempt(attempt: RecoveryAttempt): void {
   313→   313→   313→    this.status = {
   314→   314→   314→      ...this.status,
   315→   315→   315→      attempts: [attempt, ...this.status.attempts].slice(0, 50),
   316→   316→   316→    };
   317→   317→   317→    this.setStatus(this.status);
   318→   318→   318→
   319→   319→   319→    // Best-effort report to server (don't block on it)
   320→   320→   320→    if (typeof fetch !== "undefined") {
   321→   321→   321→      fetch("/api/emergency/report", {
   322→   322→   322→        method: "POST",
   323→   323→   323→        headers: { "Content-Type": "application/json" },
   324→   324→   324→        body: JSON.stringify({
   325→   325→   325→          planId: attempt.planId,
   326→   326→   326→          attempt: {
   327→   327→   327→            candidateId: attempt.candidateId,
   328→   328→   328→            candidateLabel: attempt.candidateLabel,
   329→   329→   329→            attemptNumber: attempt.attemptNumber,
   330→   330→   330→            state: attempt.state,
   331→   331→   331→            result: attempt.result,
   332→   332→   332→            reason: attempt.reason,
   333→   333→   333→            latencyMs: attempt.latencyMs,
   334→   334→   334→            startedAt: attempt.startedAt,
   335→   335→   335→            endedAt: attempt.endedAt,
   336→   336→   336→          },
   337→   337→   337→        }),
   338→   338→   338→      }).catch(() => {
   339→   339→   339→        // Network may be down — that's OK, we keep the local log
   340→   340→   340→      });
   341→   341→   341→    }
   342→   342→   342→  }
   343→   343→   343→
   344→   344→   344→  // The main recovery cycle. Returns the final status.
   345→   345→   345→  async recover(): Promise<RecoveryStatus> {
   346→   346→   346→    // Set up abort signal
   347→   347→   347→    this.abortController = new AbortController();
   348→   348→   348→
   349→   349→   349→    // 1. Start → diagnosing
   350→   350→   350→    this.emit({ type: "start" });
   351→   351→   351→
   352→   352→   352→    // 2. Diagnose the network (real client-side probes)
   353→   353→   353→    const diagnosis = await diagnoseClientSide();
   354→   354→   354→
   355→   355→   355→    // 3. If no base network, fail closed immediately
   356→   356→   356→    if (diagnosis.baseNetwork === "offline") {
   357→   357→   357→      this.emit({
   358→   358→   358→        type: "no-base-network",
   359→   359→   359→        reason: "Device is offline (navigator.onLine=false). No network interface is active. Connect to Wi-Fi or mobile data, then retry.",
   360→   360→   360→      });
   361→   361→   361→      this.logAttempt({
   362→   362→   362→        planId: "no-plan",
   363→   363→   363→        candidateId: "none",
   364→   364→   364→        candidateLabel: "Network diagnosis",
   365→   365→   365→        attemptNumber: 1,
   366→   366→   366→        state: "exhausted",
   367→   367→   367→        result: "failure",
   368→   368→   368→        reason: diagnosis.summary,
   369→   369→   369→        latencyMs: null,
   370→   370→   370→        startedAt: new Date().toISOString(),
   371→   371→   371→        endedAt: new Date().toISOString(),
   372→   372→   372→      });
   373→   373→   373→      return this.getStatus();
   374→   374→   374→    }
   375→   375→   375→
   376→   376→   376→    // 4. Gather candidates: local configs first, then remote (if fetchable)
   377→   377→   377→    const candidates = await this.gatherCandidates(diagnosis);
   378→   378→   378→
   379→   379→   379→    if (candidates.length === 0) {
   380→   380→   380→      this.emit({
   381→   381→   381→        type: "no-candidates-left",
   382→   382→   382→        reason: "No VPN configs available. Import a config from the Community Mesh tab or paste your own. Without at least one config, the recovery engine has nothing to try.",
   383→   383→   383→      });
   384→   384→   384→      this.logAttempt({
   385→   385→   385→        planId: "no-plan",
   386→   386→   386→        candidateId: "none",
   387→   387→   387→        candidateLabel: "Candidate gathering",
   388→   388→   388→        attemptNumber: 1,
   389→   389→   389→        state: "exhausted",
   390→   390→   390→        result: "failure",
   391→   391→   391→        reason: "No candidates available locally or remotely.",
   392→   392→   392→        latencyMs: null,
   393→   393→   393→        startedAt: new Date().toISOString(),
   394→   394→   394→        endedAt: new Date().toISOString(),
   395→   395→   395→      });
   396→   396→   396→      return this.getStatus();
   397→   397→   397→    }
   398→   398→   398→
   399→   399→   399→    // 5. Build the plan (client-side — we rank locally)
   400→   400→   400→    const plan: RecoveryPlan = {
   401→   401→   401→      planId: `plan_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   402→   402→   402→      diagnosis,
   403→   403→   403→      candidates,
   404→   404→   404→      policy: {
   405→   405→   405→        maxAttempts: 10,
   406→   406→   406→        verifyTimeoutMs: 8000,
   407→   407→   407→        baseDelayMs: 1500,
   408→   408→   408→        maxDelayMs: 30000,
   409→   409→   409→        maxConsecutiveFailures: 5,
   410→   410→   410→        autoRetryAfterExhaustion: false,
   411→   411→   411→        enforceKillSwitch: true,
   412→   412→   412→        requireDnsLeakProtection: true,
   413→   413→   413→      },
   414→   414→   414→      rationale: `Client-side plan: ${candidates.length} candidates. Base network: ${diagnosis.baseNetwork}.`,
   415→   415→   415→      confidence: candidates.length >= 5 ? "medium" : "low",
   416→   416→   416→      createdAt: new Date().toISOString(),
   417→   417→   417→      userTier: "free", // determined client-side; server verifies
   418→   418→   418→    };
   419→   419→   419→
   420→   420→   420→    this.status = { ...this.status, activePlan: plan, attemptsThisCycle: 0 };
   421→   421→   421→    this.setStatus(this.status);
   422→   422→   422→
   423→   423→   423→    // 6. Diagnosis complete → selecting
   424→   424→   424→    this.emit({ type: "diagnosis-complete" });
   425→   425→   425→
   426→   426→   426→    // 7. Try each candidate in sequence
   427→   427→   427→    for (let i = 0; i < plan.candidates.length; i++) {
   428→   428→   428→      if (this.abortController?.signal.aborted) break;
   429→   429→   429→
   430→   430→   430→      const candidate = plan.candidates[i];
   431→   431→   431→      const attemptNumber = i + 1;
   432→   432→   432→
   433→   433→   433→      // Select this candidate
   434→   434→   434→      this.emit({ type: "candidate-selected", candidate });
   435→   435→   435→
   436→   436→   436→      // connecting
   437→   437→   437→      const attemptStart = new Date().toISOString();
   438→   438→   438→
   439→   439→   439→      // If the engine can't connect, fail honestly
   440→   440→   440→      if (!this.engine || !this.engine.canConnect) {
   441→   441→   441→        const reason = this.engine
   442→   442→   442→          ? `Engine "${this.engine.platform}" cannot establish tunnels in this context.`
   443→   443→   443→          : "No tunnel engine available.";
   444→   444→   444→
   445→   445→   445→        this.emit({ type: "engine-failed", reason });
   446→   446→   446→
   447→   447→   447→        this.logAttempt({
   448→   448→   448→          planId: plan.planId,
   449→   449→   449→          candidateId: candidate.id,
   450→   450→   450→          candidateLabel: candidate.label,
   451→   451→   451→          attemptNumber,
   452→   452→   452→          state: "recovering",
   453→   453→   453→          result: "failure",
   454→   454→   454→          reason,
   455→   455→   455→          latencyMs: null,
   456→   456→   456→          startedAt: attemptStart,
   457→   457→   457→          endedAt: new Date().toISOString(),
   458→   458→   458→        });
   459→   459→   459→
   460→   460→   460→        await markFailed(candidate.id);
   461→   461→   461→
   462→   462→   462→        // Try next candidate
   463→   463→   463→        const next = plan.candidates[i + 1];
   464→   464→   464→        if (next) {
   465→   465→   465→          // Backoff before next try
   466→   466→   466→          await this.delay(1500);
   467→   467→   467→          this.emit({ type: "next-candidate-available", candidate: next });
   468→   468→   468→          continue;
   469→   469→   469→        } else {
   470→   470→   470→          this.emit({
   471→   471→   471→            type: "no-candidates-left",
   472→   472→   472→            reason: `All ${plan.candidates.length} candidates tried. Engine "${this.engine?.platform}" cannot establish tunnels. Install the Distort Desktop or Android app for real connectivity.`,
   473→   473→   473→          });
   474→   474→   474→          break;
   475→   475→   475→        }
   476→   476→   476→      }
   477→   477→   477→
   478→   478→   478→      // Engine CAN connect — try it
   479→   479→   479→      const result = await this.engine.connect(candidate.configUri, plan.policy.verifyTimeoutMs);
   480→   480→   480→
   481→   481→   481→      if (result.started && result.trafficVerified && result.exitIp !== result.realIp) {
   482→   482→   482→        // SUCCESS — tunnel verified, traffic flows, exit IP differs from real IP
   483→   483→   483→        this.emit({
   484→   484→   484→          type: "traffic-verified",
   485→   485→   485→          exitIp: result.exitIp,
   486→   486→   486→          exitCountry: null, // could do a lookup later
   487→   487→   487→        });
   488→   488→   488→
   489→   489→   489→        await markVerified(candidate.id);
   490→   490→   490→
   491→   491→   491→        this.logAttempt({
   492→   492→   492→          planId: plan.planId,
   493→   493→   493→          candidateId: candidate.id,
   494→   494→   494→          candidateLabel: candidate.label,
   495→   495→   495→          attemptNumber,
   496→   496→   496→          state: "connected",
   497→   497→   497→          result: "success",
   498→   498→   498→          reason: `Tunnel verified. Exit IP: ${result.exitIp}. Latency: ${result.latencyMs}ms.`,
   499→   499→   499→          latencyMs: result.latencyMs,
   500→   500→   500→          startedAt: attemptStart,
   501→   501→   501→          endedAt: new Date().toISOString(),
   502→   502→   502→        });
   503→   503→   503→
   504→   504→   504→        // Start health monitoring
   505→   505→   505→        this.startHealthMonitoring();
   506→   506→   506→        return this.getStatus();
   507→   507→   507→      }
   508→   508→   508→
   509→   509→   509→      // FAILURE — either engine didn't start, traffic didn't verify, or leak detected
   510→   510→   510→      let reason: string;
   511→   511→   511→      if (!result.started) {
   512→   512→   512→        reason = result.error || "Engine failed to start.";
   513→   513→   513→      } else if (result.exitIp === result.realIp && result.exitIp !== null) {
   514→   514→   514→        reason = `Traffic leak detected: exit IP (${result.exitIp}) equals real IP. Tunnel is not actually routing traffic.`;
   515→   515→   515→      } else if (!result.trafficVerified) {
   516→   516→   516→        reason = result.error || "Tunnel started but HTTP verification failed — server may be unreachable or credentials invalid.";
   517→   517→   517→      } else {
   518→   518→   518→        reason = result.error || "Unknown failure.";
   519→   519→   519→      }
   520→   520→   520→
   521→   521→   521→      this.emit({ type: "verification-failed", reason });
   522→   522→   522→
   523→   523→   523→      await markFailed(candidate.id);
   524→   524→   524→
   525→   525→   525→      this.logAttempt({
   526→   526→   526→        planId: plan.planId,
   527→   527→   527→        candidateId: candidate.id,
   528→   528→   528→        candidateLabel: candidate.label,
   529→   529→   529→        attemptNumber,
   530→   530→   530→        state: "recovering",
   531→   531→   531→        result: "failure",
   532→   532→   532→        reason,
   533→   533→   533→        latencyMs: result.latencyMs,
   534→   534→   534→        startedAt: attemptStart,
   535→   535→   535→        endedAt: new Date().toISOString(),
   536→   536→   536→      });
   537→   537→   537→
   538→   538→   538→      // Check exhaustion
   539→   539→   539→      this.status = {
   540→   540→   540→        ...this.status,
   541→   541→   541→        attemptsThisCycle: attemptNumber,
   542→   542→   542→      };
   543→   543→   543→      if (attemptNumber >= plan.policy.maxAttempts) {
   544→   544→   544→        this.emit({
   545→   545→   545→          type: "no-candidates-left",
   546→   546→   546→          reason: `Max attempts (${plan.policy.maxAttempts}) reached. ${attemptNumber} candidates tried, none verified. Last failure: ${reason}`,
   547→   547→   547→        });
   548→   548→   548→        break;
   549→   549→   549→      }
   550→   550→   550→
   551→   551→   551→      // Try next candidate
   552→   552→   552→      const next = plan.candidates[i + 1];
   553→   553→   553→      if (next) {
   554→   554→   554→        // Exponential backoff
   555→   555→   555→        const delayMs = Math.min(
   556→   556→   556→          plan.policy.baseDelayMs * Math.pow(2, attemptNumber - 1),
   557→   557→   557→          plan.policy.maxDelayMs
   558→   558→   558→        );
   559→   559→   559→        await this.delay(delayMs);
   560→   560→   560→        this.emit({ type: "next-candidate-available", candidate: next });
   561→   561→   561→      } else {
   562→   562→   562→        this.emit({
   563→   563→   563→          type: "no-candidates-left",
   564→   564→   564→          reason: `All ${plan.candidates.length} candidates exhausted. None verified tunnel traffic. Install the Distort Desktop or Android app for real connectivity, or import fresh configs from the Community Mesh.`,
   565→   565→   565→        });
   566→   566→   566→        break;
   567→   567→   567→      }
   568→   568→   568→    }
   569→   569→   569→
   570→   570→   570→    return this.getStatus();
   571→   571→   571→  }

   573→   573→   573→  // Gather candidates: local configs first (offline-available), then remote.
   574→   574→   574→  private async gatherCandidates(diagnosis: NetworkDiagnosis): Promise<CandidatePath[]> {
   575→   575→   575→    const candidates: CandidatePath[] = [];
   576→   576→   576→
   577→   577→   577→    // 1. Local configs (from IndexedDB — available offline)
   578→   578→   578→    try {
   579→   579→   579→      const local = await getAllConfigs();
   580→   580→   580→      // Sort: verified first, then by score
   581→   581→   581→      const sorted = [...local].sort((a, b) => {
   582→   582→   582→        if (a.lastVerifiedAt && !b.lastVerifiedAt) return -1;
   583→   583→   583→        if (!a.lastVerifiedAt && b.lastVerifiedAt) return 1;
   584→   584→   584→        return b.score - a.score;
   585→   585→   585→      });
   586→   586→   586→      for (const cfg of sorted) {
   587→   587→   587→        candidates.push(this.storedToCandidate(cfg));
   588→   588→   588→      }
   589→   589→   589→    } catch {
   590→   590→   590→      // IndexedDB not available — skip local configs
   591→   591→   591→    }
   592→   592→   592→
   593→   593→   593→    // 2. Remote configs (from API — only if internet is up)
   594→   594→   594→    if (this.fetchRemote && diagnosis.internationalReachable) {
   595→   595→   595→      try {
   596→   596→   596→        const resp = await fetch("/api/imported-configs?bucket=optimized&limit=20");
   597→   597→   597→        if (resp.ok) {
   598→   598→   598→          const data = await resp.json() as { configs: Array<{
   599→   599→   599→            id: string;
   600→   600→   600→            protocol: string;
   601→   601→   601→            uri: string;
   602→   602→   602→            label: string | null;
   603→   603→   603→            host: string;
   604→   604→   604→            port: number;
   605→   605→   605→            score: number;
   606→   606→   606→            flags: string;
   607→   607→   607→            country: string | null;
   608→   608→   608→          }> };
   609→   609→   609→          for (const cfg of data.configs) {
   610→   610→   610→            // Don't duplicate configs we already have locally
   611→   611→   611→            if (candidates.some((c) => c.id === cfg.id)) continue;
   612→   612→   612→            const flagsArray = cfg.flags ? cfg.flags.split(",").filter(Boolean) : [];
   613→   613→   613→            candidates.push({
   614→   614→   614→              id: cfg.id,
   615→   615→   615→              label: cfg.label || `${cfg.protocol} ${cfg.host}:${cfg.port}`,
   616→   616→   616→              protocol: cfg.protocol as TransportType,
   617→   617→   617→              host: cfg.host,
   618→   618→   618→              port: cfg.port,
   619→   619→   619→              stealth: flagsArray.some((f) => ["Reality", "TLS", "SNI", "Vision"].includes(f)),
   620→   620→   620→              antiDpiScore: cfg.score,
   621→   621→   621→              exitCountry: cfg.country,
   622→   622→   622→              vipOnly: false,
   623→   623→   623→              configUri: cfg.uri,
   624→   624→   624→              lastProbe: null,
   625→   625→   625→              successStreak: 0,
   626→   626→   626→              failureStreak: 0,
   627→   627→   627→              lastTriedAt: null,
   628→   628→   628→            });
   629→   629→   629→          }
   630→   630→   630→        }
   631→   631→   631→      } catch {
   632→   632→   632→        // API unreachable — fall back to local only
   633→   633→   633→      }
   634→   634→   634→    }
   635→   635→   635→
   636→   636→   636→    return candidates;
   637→   637→   637→  }

   639→   639→   639→  private storedToCandidate(cfg: StoredConfig): CandidatePath {

   645→   645→   645→        ? {

   654→   654→   654→            failureReason: null,

   660→   660→   660→  // Health monitoring — runs every 30s after connected.
   661→   661→   661→  // Detects quality degradation and triggers recovery if needed.
   662→   662→   662→  private startHealthMonitoring(): void {
   663→   663→   663→    this.monitoringInterval = setInterval(async () => {
   664→   664→   664→      if (this.status.state !== "connected" && this.status.state !== "degraded") {
   665→   665→   665→        this.stopHealthMonitoring();
   666→   666→   666→        return;
   667→   667→   667→      }
   668→   668→   668→      const health = await quickHealthCheck(3000);
   669→   669→   669→      if (!health.online) {
   670→   670→   670→        // Network dropped — trigger recovery
   671→   671→   671→        this.emit({ type: "quality-dropped" });
   672→   672→   672→        // Auto-recover after a short delay
   673→   673→   673→        setTimeout(() => {
   674→   674→   674→          if (this.status.state === "degraded") {
   675→   675→   675→            this.recover().catch(() => {});
   676→   676→   676→          }
   677→   677→   677→        }, 2000);
   678→   678→   678→      } else if (health.latencyMs !== null && health.latencyMs > 2000) {
   679→   679→   679→        // High latency — degraded
   680→   680→   680→        if (this.status.state === "connected") {
   681→   681→   681→          this.emit({ type: "quality-dropped" });
   682→   682→   682→        }
   683→   683→   683→      } else {
   684→   684→   684→        // Healthy — if we were degraded, recover
   685→   685→   685→        if (this.status.state === "degraded") {
   686→   686→   686→          this.emit({ type: "quality-recovered" });
   687→   687→   687→        }
   688→   688→   688→      }
   689→   689→   689→    }, 30000);
   690→   690→   690→
   691→   691→   691→    // Listen for network change events (Wi-Fi → mobile data switch)
   692→   692→   692→    this.networkChangeCleanup = onNetworkChange(() => {
   693→   693→   693→      // Re-diagnose on network change
   694→   694→   694→      if (this.status.state === "connected") {
   695→   695→   695→        this.emit({ type: "quality-dropped" });
   696→   696→   696→        setTimeout(() => {
   697→   697→   697→          if (this.status.state === "degraded") {
   698→   698→   698→            this.recover().catch(() => {});
   699→   699→   699→          }
   700→   700→   700→        }, 1000);
   701→   701→   701→      }
   702→   702→   702→    });
   703→   703→   703→  }
   704→   704→   704→
   705→   705→   705→  private stopHealthMonitoring(): void {
   706→   706→   706→    if (this.monitoringInterval) {
   707→   707→   707→      clearInterval(this.monitoringInterval);
   708→   708→   708→      this.monitoringInterval = null;
   709→   709→   709→    }
   710→   710→   710→    if (this.networkChangeCleanup) {
   711→   711→   711→      this.networkChangeCleanup();
   712→   712→   712→      this.networkChangeCleanup = null;
   713→   713→   713→    }
   714→   714→   714→  }

   716→   716→   716→  private delay(ms: number): Promise<void> {
   717→   717→   717→    return new Promise((resolve) => {
   718→   718→   718→      setTimeout(resolve, Math.min(ms, 30000));
   719→   719→   719→    });
   720→   720→   720→  }

   722→   722→   722→  // Manual disconnect
   723→   723→   723→  async disconnect(): Promise<void> {
   724→   724→   724→    this.stopHealthMonitoring();
   725→   725→   725→    if (this.engine) {
   726→   726→   726→      await this.engine.disconnect();
   727→   727→   727→    }
   728→   728→   728→    this.emit({ type: "disconnect" });
   729→   729→   729→  }
